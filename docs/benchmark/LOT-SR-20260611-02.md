04 // ALERT LEVELS (DOCTRINE)
--------------------------------------------------------------------------------

  [BRAVO]   OVERDUE +{N}D — {TYPE}: {text} — {DATE}
            Fires for every entry past its scheduled date.
            Displayed in Command Board under // OVERDUE [{N}]

  [ALPHA]   {TYPE}: {text} — {DATE}
            Fires for entries scheduled today.
            Displayed in Command Board under // TODAY

  [NOTICE]  T-MINUS {N}D — {TYPE}: {text} — {DATE}
            Fires for entries 1 day out (tomorrow).
            Displayed in Command Board under // STANDBY

  [CLEARED] {TYPE}: {text} — {DATE} — COMPLETE
            Fires when user marks an entry done (× button).
            Entry is deleted from calendar; CLEARED log persists.
